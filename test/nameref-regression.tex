\directlua{
  local function check(package, label, field, expected)
    local result = docref.reference(package, label, field)
    assert(result == expected, field .. ': ' .. result .. ' expected ' .. expected)
  end
  for _, package in ipairs({'fields-aux', 'fields-pdf'}) do
    check(package, 'sec:target', 'number', '1')
    check(package, 'sec:target', 'name', 'Target code & 50' .. string.char(37))
    check(package, 'sec:target', 'page', '40')
    check(package, 'sec:preface', 'name', 'Preface')
    check(package, 'sec:preface', 'page', 'iii')
  end
  check('fields-pdf', 'sec:unicode', 'name', 'Résumé – données')
  check('fields-aux', 'sec:unicode', 'name', 'Résumé – données')
  check('fields-source', 'sec:target', 'name', 'Target code & 50' .. string.char(37))
  check('fields-aux', 'aux-only', 'name', 'عنوان فارسی')
  check('fields-aux', 'aux-only', 'page', 'iii')
  assert(not pcall(docref.reference, 'fields-source', 'sec:target', 'page'),
    'Source-only page reference fabricated a page number')
}
